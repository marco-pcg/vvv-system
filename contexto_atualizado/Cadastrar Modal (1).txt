@startuml
title Cadastrar Modal (Visão de Domínio)
hide footbox

' --- DEFINIÇÃO DE PARTICIPANTES ---
actor "Administrador" as Admin

' Entidades (Verde Claro)
entity "FrotaManager" as Manager #FFCC99
entity "Transportadora" as Transp #99FF99
entity "Modal" as Modal #99FF99

' --- INÍCIO ---
activate Admin

Admin -> Manager: 1: cadastrarModal(dadosModal)
activate Manager #FFCC99

' Validação na Transportadora
Manager -> Transp: 2: verificarExistenciaTransportadora(idTransportadora)
activate Transp #99FF99
Transp --> Manager: 3: transportadoraExiste
deactivate Transp

' Criação do Modal
Manager -> Modal: [transportadora existe] 4: <<create>> new Modal(dadosModal)
activate Modal #99FF99
Modal --> Manager: 5: modalInstanciado
deactivate Modal

' Retorno
Manager --> Admin: 6: modalCadastradoSucesso()

deactivate Manager
deactivate Admin
@enduml

Nós @startuml
title Processo de Compra
hide footbox

' --- DEFINIÇÃO DE PARTICIPANTES ---
actor "Cliente" as C

' Entidades de Domínio (Verde Claro)
entity "Viagem" as V #99FF99
entity "Modal" as Mod #99FF99
entity "Reserva" as Res #99FF99
entity "Passageiro" as Pass #99FF99
entity "Pagamento" as Pgto #99FF99
entity "Ticket" as Tkt #99FF99

' --- INÍCIO DO PROCESSO ---
activate C

C->V: 1: selecionarViagem(origem, destino, data)
activate V #99FF99

V->Mod: 2: verificarDisponibilidadeCapacidade()
activate Mod #99FF99
Mod-->V: 3: vagasDisponiveis
deactivate Mod

C->V: 4: solicitarReserva(listaPassageiros)

' CRIAÇÃO DA RESERVA
V->Res: [vagas > 0] 5: <<create>> novaReserva(Cliente, Viagem)
activate Res #99FF99

' --- LOOP PASSAGEIROS ---
Res->Pass: *[Para cada passageiro] 6: validarCadastroIdade()
activate Pass #99FF99
Pass-->Res: 7: dadosValidados
deactivate Pass

' Inclusão Desconto 
Res->Res: [idade 2 a 10 anos E com acompanhante] 8: aplicarDesconto(40%)

Res->Res: 9: calcularValorFinal()
Res-->V: 10: reservaCriada (Status: Pendente)
deactivate Res

V-->C: 11: solicitarPagamento(valorFinal)

C->Pgto: 12: selecionarEMitirPagamento(dadosFinanceiros)
activate Pgto #99FF99

Pgto->Pgto: 13: processarTransacao(parcelamento, juros)
Pgto-->V: 14: pagamentoAprovado
deactivate Pgto

' --- EMISSÃO DE TICKET ---
V->Mod: [pgto Aprovado] 15: verificarStatusOperacional()
activate Mod #99FF99
Mod-->V: 16: status="Operacional"
deactivate Mod

V->Res: 17: confirmarPagamento()
activate Res #99FF99

' Criação do Ticket
Res->Tkt: [Modal Operacional] 18: <<create>> emitirTicket()
activate Tkt #99FF99
Tkt-->Res: 19: ticketGerado
deactivate Tkt

Res->Res: 20: setStatus("Confirmada")
Res-->V: 21: reservaConfirmada
deactivate Res

V-->C: 22: entregarComprovanteTicket()

deactivate V
deactivate C
@enduml
